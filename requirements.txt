# InsurClaim AI Python dependencies
# We install packages phase by phase, so this file grows as the project grows.
# Heavy machine-learning packages (torch, transformers, ...) are commented out
# below and will be enabled when we reach the fine-tuning phase (Phase 5).

# --- Core: Claude API + configuration (Phase 1) ---
anthropic            # Claude API SDK (primary LLM)
python-dotenv        # loads API keys / secrets from the .env file
pydantic             # data models; validates structured claim JSON

# --- Multi-provider LLM fallback (Phase 1, used inside llm.py) ---
openai               # OpenRouter gateway (OpenAI-compatible) cheap Claude access in dev
google-genai         # Gemini API (fallback; supports vision) current SDK
groq                 # Groq API (fallback; very fast, text-only)

# --- Database: PostgreSQL (Phase 1) ---
psycopg[binary]      # PostgreSQL driver used by db/repo.py

# --- Orchestration: LangGraph (Phase 8) ---
langgraph            # multi-agent StateGraph that wires the 4 agents
langchain-core       # core primitives LangGraph builds on

# --- Vector DB / RAG (Phase 3) ---
chromadb               # local vector database (uses its own lightweight embeddings)
# sentence-transformers # optional stronger embeddings (pulls in torch)

# --- Image checks (Phase 4) ---
Pillow                 # read image EXIF/metadata
imagehash              # perceptual hash for duplicate-photo detection

# --- Document intake & reporting ---
pypdf                  # extract text from uploaded PDF claim documents (fallback)
pdfplumber             # stronger PDF text extraction (tried before pypdf)
pypdfium2              # rasterize scanned PDF pages for vision OCR fallback
fpdf2                  # render the downloadable PDF investigation report

# --- Multi-format intake (documents + video) ---
python-docx            # extract text from uploaded Word (.docx) documents
opencv-python-headless # sample frames from uploaded videos (dashcam / walkaround)

# --- Frontend dashboard (Phase 11) ---
streamlit              # investigator dashboard UI (frontend/app.py)

# --- API backend for the React dashboard (api/main.py) ---
starlette              # lightweight ASGI framework (already pulled in by streamlit)
uvicorn                # ASGI server that runs the API (python -m uvicorn api.main:app)
python-multipart       # multipart/form-data parsing for file uploads

# --- Evaluation (Phase 9) ---
scikit-learn           # precision / recall / confusion matrix
numpy                  # numeric helpers
pandas                 # tabular handling for eval

# --- Fine-tuning (Phase 5) ---
torch                 # deep-learning backend (CPU build on Windows by default)
transformers          # HuggingFace models (DistilBERT sequence classifier)
accelerate            # required backend for the HuggingFace Trainer
datasets              # dataset handling for training
# peft                 # LoRA fine-tuning not used yet (train.py does full fine-tuning)
